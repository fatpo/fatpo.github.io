## 2. python脚本

```python3
import os
import subprocess

import requests


def main():
    # 下载文件列表
    file_list = "file_list.txt"

    # 清空文件列表
    with open(file_list, 'w') as f:
        pass

    index = 0
    while True:
        file_name = f"https://xxx.com/_video_{index}.ts?zbid=111&tpid=222"
        url = f"{base_url}{index}{suffix}"

        print(f"Downloading {file_name} from {url}...")
        response = requests.get(url)

        if response.status_code != 200 or not response.content:
            print("No more files to download. Stopping.")
            break

        # 检查是否包含错误信息
        if response.content.startswith(b"<?xml"):
            print("Received error XML. Stopping.")
            break

        with open(file_name, 'wb') as f:
            f.write(response.content)

        with open(file_list, 'a') as f:
            f.write(f"file '{file_name}'\n")

        index += 1

    # 合并文件
    print("Merging files into output.mp4...")
    subprocess.run(['ffmpeg', '-f', 'concat', '-safe', '0', '-i', file_list, '-c', 'copy', 'output.mp4'])

    # 清理
    print("Cleaning up...")
    for i in range(30, index):
        os.remove(f"video_{i}.ts")
    os.remove(file_list)

    print("Done.")


if __name__ == '__main__':
    main()

```

## 3. 原理

现在的在线视频都是一个ts文件接着一个ts下载，然后视频网站不断的解析。

我们可以把所有的ts先拉下来，再用 ffmpeg 合并成一个mp4文件。
